SELECT 
p.nombre AS producto,
p.precio,
c.nombre AS categoria,
pr.nombre AS proveedor
FROM productos p
JOIN categoria c ON p.categoria_fk = c.id
JOIN proveedores pr ON p.proveedor_fk = pr.id;


SELECT 
p.id,
p.nombre AS producto,
COUNT(dp.pedido_fk) AS cantidad_pedidos
FROM productos p
JOIN detalle_pedido dp ON p.id = dp.producto_fk
GROUP BY p.id, p.nombre
ORDER BY cantidad_pedidos DESC;

SELECT 
c.id,
CONCAT(c.nombre, ' ', c.apellido) AS cliente,
COUNT(p.id) AS total_pedidos
FROM clientes c
LEFT JOIN pedido p ON c.id = p.clientes_fk
GROUP BY c.id, c.nombre, c.apellido;

SELECT 
CONCAT(c.nombre, ' ', c.apellido) AS nombre_completo,
COUNT(p.id) AS cantidad_pedidos,
SUM(p.total) AS total_facturado
FROM clientes c
JOIN pedido p ON c.id = p.clientes_fk
GROUP BY c.id, c.nombre, c.apellido
HAVING SUM(p.total) > 500000
ORDER BY total_facturado DESC;


SELECT 
productos.nombre AS Nombre_del_producto,
productos.precio AS Precio,
categoria.nombre AS Categoría,
proveedores.nombre AS Proveedor
FROM productos
JOIN categoria ON productos.categoria_fk = categoria.id
JOIN proveedores ON productos.proveedor_fk = proveedores.id
LEFT JOIN detalle_pedido ON productos.id = detalle_pedido.producto_fk
WHERE detalle_pedido.producto_fk IS NULL;
